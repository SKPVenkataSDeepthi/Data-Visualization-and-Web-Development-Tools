Label each point on the scatter plot using the text elements. The text of the label should be the two values separated by a comma and a space. For example, the label for the first point is 34, 78. Set the x attribute so it's 5 units more than the value you used for the cx attribute on the circle. Set the y attribute the same way that's used for the cy value on the circle.



<body>
  <script>
    const dataset = [
      [34, 78],
      [109, 280],
      [310, 120],
      [79, 411],
      [420, 220],
      [233, 145],
      [333, 96],
      [222, 333],
      [78, 320],
      [21, 123]
    ];

    const w = 500;
    const h = 500;

    const svg = d3.select("body")
                  .append("svg")
                  .attr("width", w)
                  .attr("height", h);

    svg.selectAll("circle")
       .data(dataset)
       .enter()
       .append("circle")
       .attr("cx", (d) => d[0])
       .attr("cy", (d) => h - d[1]) // Adjust cy to display the chart right-side-up
       .attr("r", 5)
       .attr("class", "bar"); // Added class bar to each rect element

    // Append text labels to the svg
    svg.selectAll("text")
       .data(dataset)
       .enter()
       .append("text")
       .attr("x", (d) => d[0] + 5) // Set x to 5 units more than cx
       .attr("y", (d) => h - d[1]) // Set y the same as cy
       .text((d) => `${d[0]}, ${d[1]}`) // Set label to the data values separated by a comma and a space
       .style("font-size", "12px")
       .style("fill", "black");
  </script>
</body>
